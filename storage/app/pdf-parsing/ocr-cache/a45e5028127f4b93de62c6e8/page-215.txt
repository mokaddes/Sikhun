CREATE TABLE student_info (roll INTEGER, name TEXT;
CREATE TABLE result (roll INTEGER, subject TEXT, marks REAL;

এই টেবিল দুটির মধ্যে roll কলাম দিয়ে একটি রিলেশন দেখা যাচ্ছে। student_info টেবিলে প্রতিটি
শিক্ষার্থীর roll ও name রয়েছে। আবার result টেবিলে, শিক্ষার্থীর রোল নম্বর ও বিভিন্ন বিষয়ে পরীক্ষায়
প্রাপ্ত নম্বর (marks) রয়েছে। student_info টেবিলের সঙ্গে result টেবিলের রিলেশন হচ্ছে ওয়ান টু
মেনি রিলেশন।
